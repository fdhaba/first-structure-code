#include <iostream>
#include <string>
using namespace std;

struct Student {

    int id;
    string name;
    double cgpa;

};

int main() {

    Student students[3] = {

        {101, "Abebe", 3.5},
        {102, "Kebede", 3.8},
        {103, "Mekdes", 3.9}

    };

    for (int i = 0; i < 3; i++) {

        cout << "Student " << i + 1 << endl;

        cout << "ID: " << students[i].id << endl;

        cout << "Name: " << students[i].name << endl;

        cout << "CGPA: " << students[i].cgpa << endl;

        cout << endl;
    }

    return 0;
}
